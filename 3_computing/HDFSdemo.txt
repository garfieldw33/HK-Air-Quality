from pyspark.sql.functions import avg

# Verify Spark sees HDFS
sc._jsc.hadoopConfiguration().get("fs.defaultFS")

# Read AQHI data from HDFS
df = spark.read.parquet("hdfs://localhost:9001/aqhi/raw/aqhi_clean_long.parquet")

print(df.count())
df.printSchema()
df.show(5)

# Station-level AQHI ranking
station_mean = (
    df.groupBy("station")
      .agg(avg("aqhi").alias("mean_aqhi"))
      .orderBy("mean_aqhi", ascending=False)
)

station_mean.show()

# Save results back to HDFS
station_mean.write.mode("overwrite").csv(
    "hdfs://localhost:9001/aqhi/output/station_mean"
)

exit()
